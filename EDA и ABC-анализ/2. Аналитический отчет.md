```python
def plot_mosaic(plot_functions, layout, figsize=(18, 14)):
    fig, ax = plt.subplot_mosaic(
        layout,
        figsize=figsize
    )

    results = {}

    for key, plot_func in plot_functions.items():
        results[key] = plot_func(ax[key])

    plt.tight_layout()
    plt.show()

    return results
```

**Функция ABC-классификации**

Для каждого показателя подкатегории сортируются по убыванию значения. Затем рассчитываются доля и накопительная доля показателя. Группы определяются по порогам: A — до 80%, B — до 95%, C — оставшаяся часть.

```python
def perform_abc(df, index):
    cols = list(df.columns)
    cols.remove(index)

    grouped_df = df.groupby(index).agg({
        col: "sum" for col in cols
    })

    for col in cols:
        grouped_df[f"rel_{col}"] = grouped_df[col] / sum(grouped_df[col])
        grouped_df = grouped_df.sort_values(
            f"rel_{col}",
            ascending=False
        )
        grouped_df[f"cumsum_{col}"] = grouped_df[f"rel_{col}"].cumsum()
        grouped_df[f"abc_{col}"] = np.where(
            grouped_df[f"cumsum_{col}"] < 0.8,
            "A",
            np.where(grouped_df[f"cumsum_{col}"] < 0.95, "B", "C")
        )

    abc_cols = [f"abc_{col}" for col in cols]
    return grouped_df[cols + abc_cols]


abc_res_df = perform_abc(abc_df, "level2").reset_index()
```
